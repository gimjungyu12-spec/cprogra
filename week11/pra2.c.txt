
#include <stdio.h>

void swap(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

int main(void)
{
    int num1, num2;

    printf("두 정수를 입력하시오 : ");
    scanf("%d %d", &num1, &num2);

    printf("교환 전 : %d %d\n", num1, num2);

    swap(&num1, &num2);

    printf("교환 후 : %d %d\n", num1, num2);

    return 0;
}
